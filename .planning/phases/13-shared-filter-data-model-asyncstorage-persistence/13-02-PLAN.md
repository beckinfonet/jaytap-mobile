```typescript
const LANGUAGE_STORAGE_KEY = '@jaytap_language';                      // → @jaytap_filter_style
export type Language = 'en' | 'ru';                                   // → FilterStyle = 'guided' | 'cascading' | 'master' | 'sentence'

interface LanguageContextType {                                       // → FilterStyleContextType
  language: Language;                                                 // → filterStyle: FilterStyle
  setLanguage: (lang: Language) => Promise<void>;                     // → setFilterStyle: (s: FilterStyle) => Promise<void>
  t: (key: TranslationKeys, params?: Record<string, string>) => string;  // → NOT MIRRORED (no translation helper for filter style)
}

const LanguageContext = createContext<LanguageContextType | undefined>(undefined);

export const LanguageProvider = ({ children }: { children: ReactNode }) => {
  const [language, setLanguageState] = useState<Language>('en');      // → ('guided')

  useEffect(() => { loadLanguage(); }, []);

  const loadLanguage = async () => {
    try {
      const stored = await AsyncStorage.getItem(LANGUAGE_STORAGE_KEY);
      if (stored === 'en' || stored === 'ru') {                       // → guarded against the 4 FilterStyle values
        setLanguageState(stored);
      }
    } catch (e) { console.error('Failed to load language preference', e); }
  };

  const setLanguage = async (lang: Language) => {
    try {
      await AsyncStorage.setItem(LANGUAGE_STORAGE_KEY, lang);
      setLanguageState(lang);
    } catch (e) { console.error('Failed to save language preference', e); }
  };

  // t helper omitted in FilterStyleContext — no translation surface

  return (
    <LanguageContext.Provider value={{ language, setLanguage, t }}>{children}</LanguageContext.Provider>
  );
};

export const useLanguage = () => {
  const ctx = useContext(LanguageContext);
  if (!ctx) throw new Error('useLanguage must be used within LanguageProvider');
  return ctx;
};
```

## Trust Boundaries

| Boundary | Description |

| Device AsyncStorage ↔ in-memory hook state | The only external surface in Plan 02. | Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-13-02-01 | Tampering | AsyncStorage value at key `@jaytap_filter_style` | mitigate | LOW severity — hand-edited storage or legacy migration noise could write a non-allowlisted string.
